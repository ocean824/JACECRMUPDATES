5) WEBHOOK SENDER CODE (PASTE THIS INTO REPO)

Create: server/services/bondaiWebhook.ts (or matching folder)

import crypto from "crypto";
import pRetry from "p-retry";

type BondAIWebhookResult = {
  ok: boolean;
  status: number;
  responseText: string;
};

function requireEnv(name: string): string {
  const v = process.env[name];
  if (!v) throw new Error(`Missing env: ${name}`);
  return v;
}

export function buildBondAISignature(secret: string, timestampMs: number, rawBody: string) {
  // signature base = "<timestamp>.<rawBody>"
  const base = `${timestampMs}.${rawBody}`;
  const hmac = crypto.createHmac("sha256", secret).update(base).digest("hex");
  return `v1=${hmac}`;
}

export async function sendBondAIWebhook(args: {
  event: string; // "settlement.confirmed"
  eventKey: string; // idempotency key
  payload: unknown;
}): Promise<BondAIWebhookResult> {
  const enabled = (process.env.BONDAI_WEBHOOK_ENABLED ?? "true").toLowerCase() === "true";
  if (!enabled) return { ok: true, status: 204, responseText: "BONDAI_WEBHOOK_ENABLED=false (skipped)" };

  const url = requireEnv("BONDAI_WEBHOOK_URL");
  const secret = requireEnv("BONDAI_WEBHOOK_SECRET");
  const timeoutMs = Number(process.env.BONDAI_WEBHOOK_TIMEOUT_MS ?? "8000");

  const timestampMs = Date.now();
  const rawBody = JSON.stringify(args.payload);
  const signature = buildBondAISignature(secret, timestampMs, rawBody);

  const attempt = async (): Promise<BondAIWebhookResult> => {
    const ac = new AbortController();
    const t = setTimeout(() => ac.abort(), timeoutMs);

    try {
      const res = await fetch(url, {
        method: "POST",
        headers: {
          "content-type": "application/json",
          "x-bondai-event": args.event,
          "x-bondai-timestamp": String(timestampMs),
          "x-bondai-signature": signature,
          "x-idempotency-key": args.eventKey,
        },
        body: rawBody,
        signal: ac.signal,
      });

      const text = await res.text();
      // Retry rules:
      // - retry on 429 and 5xx
      // - do not retry on other 4xx
      if (res.status === 429 || (res.status >= 500 && res.status <= 599)) {
        throw new pRetry.AbortError(`Retryable status ${res.status}: ${text.slice(0, 300)}`);
      }
      if (res.status >= 400) {
        // non-retryable client error
        return { ok: false, status: res.status, responseText: text };
      }
      return { ok: true, status: res.status, responseText: text };
    } finally {
      clearTimeout(t);
    }
  };

  // p-retry: retry transient failures (network/abort/429/5xx)
  const result = await pRetry(attempt, {
    retries: Number(process.env.BONDAI_WEBHOOK_MAX_ATTEMPTS ?? "8"),
    factor: 2,
    minTimeout: 500,
    maxTimeout: 10_000,
    randomize: true,
  }).catch((err: any) => {
    // If it exhausts retries or hits AbortError, treat as failed delivery attempt
    return { ok: false, status: 0, responseText: String(err?.message ?? err) } as BondAIWebhookResult;
  });

  return result;
}

6) OUTBOX DISPATCHER (PASTE THIS TOO)

Create: server/jobs/webhookOutboxDispatcher.ts (or similar)

import { sendBondAIWebhook } from "../services/bondaiWebhook";

// These types/functions must match your repo’s DB layer.
// Replace with Drizzle/Prisma SQL calls accordingly.
import { db } from "../db";
import { webhookOutbox } from "../db/schema"; // adjust path

const DISPATCH_INTERVAL_MS = 10_000;
let started = false;

export function startWebhookOutboxDispatcher() {
  if (started) return;
  started = true;

  setInterval(async () => {
    try {
      // Fetch due events (limit small)
      const due = await db.query.webhookOutbox.findMany({
        where: (t: any, { and, eq, lte }: any) =>
          and(eq(t.status, "PENDING"), lte(t.nextAttemptAt, new Date())),
        limit: 10,
        orderBy: (t: any, { asc }: any) => [asc(t.nextAttemptAt)],
      });

      for (const row of due) {
        if (row.destination !== "BONDAI") continue;

        const result = await sendBondAIWebhook({
          event: row.eventType,
          eventKey: row.eventKey,
          payload: row.payloadJson,
        });

        if (result.ok) {
          await db.update(webhookOutbox).set({
            status: "DELIVERED",
            attemptCount: row.attemptCount + 1,
            lastStatusCode: result.status,
            lastError: null,
            updatedAt: new Date(),
          }).where((t: any, { eq }: any) => eq(t.id, row.id));
        } else {
          const attemptCount = row.attemptCount + 1;
          const maxAttempts = Number(process.env.BONDAI_WEBHOOK_MAX_ATTEMPTS ?? "8");

          // Backoff schedule
          const backoffMs = Math.min(60_000, 1_000 * Math.pow(2, Math.min(attemptCount, 6)));
          const nextAttempt = new Date(Date.now() + backoffMs);

          await db.update(webhookOutbox).set({
            status: attemptCount >= maxAttempts ? "FAILED" : "PENDING",
            attemptCount,
            lastStatusCode: result.status || null,
            lastError: result.responseText?.slice(0, 1000) ?? "unknown error",
            nextAttemptAt: attemptCount >= maxAttempts ? null : nextAttempt,
            updatedAt: new Date(),
          }).where((t: any, { eq }: any) => eq(t.id, row.id));
        }
      }
    } catch (err) {
      // log but never crash server loop
      console.error("[webhookOutboxDispatcher] error:", err);
    }
  }, DISPATCH_INTERVAL_MS);
}


Then call startWebhookOutboxDispatcher() from server startup (e.g., server/index.ts).
