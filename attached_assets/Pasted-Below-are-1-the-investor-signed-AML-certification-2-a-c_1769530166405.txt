  ];

  return (
    <footer className="text-xs py-6 border-t">
      <nav className="flex flex-wrap gap-x-4 gap-y-2">
        {items.map((i) =>
          i.newWindow ? (
            <a key={i.href} href={i.href} target="_blank" rel="noopener noreferrer" className="underline">
              {i.label} <span className="sr-only">(Opens in a new window)</span>
            </a>
          ) : (
            <Link key={i.href} href={i.href} className="underline">
              {i.label}
            </Link>
          )
        )}
      </nav>

      <div className="mt-3 opacity-70">
        This is for persons in the US only.
      </div>
    </footer>
  );
